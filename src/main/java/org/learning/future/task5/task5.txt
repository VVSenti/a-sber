public class ReportService {

private final ExecutorService executor = Executors.newFixedThreadPool(10);

private final Map<Long, Report> reports = new HashMap<>();      -- raceCondition

public void generateReports(List<Long> ids) {

	for (Long id : ids) {
		executor.submit(() -> {
			Report report = generateReport(id);
			reports.put(id, report);

		});     -- игнорируется Future. Если вылетит ошибка, то в вызывающий поток она не пробросится
	}

	executor.shutdown();    -- вызов метода убьет executor, и тот больше не вызвать
	-- нет ожидания завершения всех потоков черещ awaitTermination

	System.out.println("Reports generated: " + reports.size());   -- если не все задачи успеют завершиться, то и не все занесуесутся в мапу.
	 остальные таски продолжат выполняться и рано или поздно обновят это значение
}
}
